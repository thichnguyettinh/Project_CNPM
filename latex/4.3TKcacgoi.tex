% 1. Macro vẽ UML 2.x Package có phần tab ở góc trên bên trái
% #1=x, #2=y, #3=w, #4=h, #5=tab_w, #6=tab_h, #7=body_fill, #8=tab_fill, #9=y_plus_h
\newcommand{\UMLPackage}[9]{%
  \ifusetikz
    \filldraw[fill=#8, draw=umlBorder, line width=0.75pt] (#1, #9) rectangle ({#1+#5}, {#9+#6});%
    \filldraw[fill=#7, draw=umlBorder, line width=0.85pt] (#1, #2) rectangle ({#1+#3}, #9);%
  \else
    \setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.75pt}%
    \put(#1,#9){\textcolor{#8}{\rule{#5cm}{#6cm}}}%
    \put(#1,#9){\textcolor{umlBorder}{\framebox(#5,#6){}}}%
    \setlength{\fboxrule}{0.85pt}%
    \put(#1,#2){\textcolor{#7}{\rule{#3cm}{#4cm}}}%
    \put(#1,#2){\textcolor{umlBorder}{\framebox(#3,#4){}}}%
  \fi
}

% 2. Macro vẽ hộp Class / Interface bên trong Package (#1=x, #2=y, #3=w, #4=h, #5=fill_color)
\newcommand{\UMLClassBox}[5]{%
  \ifusetikz
    \filldraw[fill=#5, draw=umlBorder, line width=0.65pt] (#1, #2) rectangle ({#1+#3}, {#2+#4});%
  \else
    \setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.65pt}%
    \put(#1,#2){\textcolor{#5}{\rule{#3cm}{#4cm}}}%
    \put(#1,#2){\textcolor{umlBorder}{\framebox(#3,#4){}}}%
  \fi
}

% 3. Macro vẽ đường kẻ ngang phân cách bên trong Package / Class (#1=x, #2=y, #3=len_cm, #4=len_bp)
\newcommand{\UMLHLine}[4]{%
  \ifusetikz
    \draw[draw=umlBorder!75, line width=0.5pt] (#1, #2) -- ({#1+#3}, #2);%
  \else
    \put(#1,#2){\RawPDF{q 0.5 w [] 0 d 0.35 0.42 0.50 RG 0 0 m #4 0 l S Q}}%
  \fi
}

% 4. Các Macro đặt nhãn văn bản căn trái (W), căn phải (E), căn giữa (C)
\newcommand{\UMLTextW}[3]{%
  \ifusetikz
    \node[text=umlText, anchor=west, inner sep=0pt] at (#1, #2) {#3};%
  \else
    \put(#1,#2){\makebox(0,0)[l]{\textcolor{umlText}{#3}}}%
  \fi
}
\newcommand{\UMLTextE}[3]{%
  \ifusetikz
    \node[text=umlText, anchor=east, inner sep=0pt] at (#1, #2) {#3};%
  \else
    \put(#1,#2){\makebox(0,0)[r]{\textcolor{umlText}{#3}}}%
  \fi
}
\newcommand{\UMLTextC}[3]{%
  \ifusetikz
    \node[text=umlText, anchor=center, inner sep=0pt] at (#1, #2) {#3};%
  \else
    \put(#1,#2){\makebox(0,0)[c]{\textcolor{umlText}{#3}}}%
  \fi
}

% 5. Macro đặt nhãn Stereotype trên đường nối Dependency (có nền trắng che nét đứt bên dưới)

  \ifusetikz
    \node[font=\fontsize{6.2pt}{7.2pt}\selectfont\itshape, text=umlText, fill=white, inner sep=1.2pt] at (#1, #2) {#3};%
  \else
    \put(#1,#2){\makebox(0,0)[c]{\setlength{\fboxsep}{1.2pt}\colorbox{white}{\textcolor{umlText}{\fontsize{6.2pt}{7.2pt}\selectfont\itshape #3}}}}%
  \fi
}

}
